module PechnikHubScanner
  def self.scan_model
    model = Sketchup.active_model
    entities = model.active_entities
    
    puts "=== ОТЧЕТ СКАНИРОВАНИЯ МОДЕЛИ ДЛЯ WEB-ПРОТОТИПА ==="
    puts "Файл: #{model.title.empty? ? 'Без имени' : model.title}"
    puts "Единицы измерения: #{self.get_units_system(model)}"
    puts "--------------------------------------------------"
    
    # 1. Анализ корневой структуры (Ищем модули: Мангал, Казан, Столешница)
    puts "\n[1. КОРНЕВАЯ СТРУКТУРА КОМПОНЕНТОВ И ГРУПП]:"
    root_elements = 0
    entities.each do |ent|
      if ent.is_a?(Sketchup::ComponentInstance) || ent.is_a?(Sketchup::Group)
        name = ent.name.empty? ? "<Без имени>" : ent.name
        type = ent.is_a?(Sketchup::Group) ? "Группа" : "Компонент"
        bbox = ent.bounds
        
        # Конвертируем габариты в мм для удобства
        w = (bbox.width.to_mm).round(0)
        d = (bbox.depth.to_mm).round(0)
        h = (bbox.height.to_mm).round(0)
        
        puts "-> #{type}: \"#{name}\" | Размеры (мм): Ш=#{w}, Д=#{d}, В=#{h} | Оси (X,Y,Z): #{ent.transformation.origin.to_a.map{|c| c.to_mm.round(1)}}"
        root_elements += 1
      end
    end
    puts "Всего объектов в корне сцены: #{root_elements}"
    
    # 2. Сканирование слоев/тегов (Tags)
    puts "\n[2. ИСПОЛЬЗУЕМЫЕ ТЕГИ / СЛОИ]:"
    model.layers.each do |layer|
      puts "- Тег: \"#{layer.name}\" #{layer.visible? ? '(Видимый)' : '(Скрыт)'}"
    end
    
    # 3. Глубокий аудит структуры порядовки (Ищем ключевые слова)
    puts "\n[3. АНАЛИЗ ПОРЯДОВКИ И МАТЕРИАЛОВ]:"
    brick_count = 0
    shamo_count = 0
    rows_detected = []
    
    # Рекурсивный обход всех вложенных элементов
    self.count_elements(entities, rows_detected, check_bricks = true)
    
    unless rows_detected.empty?
      puts "Обнаружены упоминания рядов в названиях: #{rows_detected.uniq.sort.join(', ')}"
    else
      puts "Внимание: В названиях групп/компонентов явных меток рядов (например, 'Ряд 1') не найдено."
    end
    
    puts "--------------------------------------------------"
    puts "Сканирование завершено. Скопируйте этот текст целиком."
  end
  
  private
  
  def self.get_units_system(model)
    options = model.options['UnitsOptions']
    case options['LengthFormat']
    when 0 then "Decimal"
    when 1 then "Architectural"
    when 2 then "Engineering"
    when 3 then "Fractional"
    else "Unknown"
    end
  end
  
  def self.count_elements(entities, rows_detected, check_bricks)
    entities.each do |ent|
      name = ent.name.downcase
      
      # Ищем упоминания рядов
      if name =~ /(ряд|row|r)\s*(\d+)/
        rows_detected << "Ряд_#{$2}"
      end
      
      # Если это вложенная группа или компонент, идем глубже
      if ent.is_a?(Sketchup::Group)
        self.count_elements(ent.entities, rows_detected, check_bricks)
      elsif ent.is_a?(Sketchup::ComponentInstance)
        self.count_elements(ent.definition.entities, rows_detected, check_bricks)
      end
    end
  end
end
